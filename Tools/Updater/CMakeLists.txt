cmake_minimum_required(VERSION 3.24)
project(Idas3UpdateInstaller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(Idas3UpdateInstaller WIN32 main.cpp)
target_compile_definitions(Idas3UpdateInstaller PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Idas3UpdateInstaller PRIVATE bcrypt shell32 user32)
if(MSVC)
  target_compile_options(Idas3UpdateInstaller PRIVATE /W4 /WX /utf-8)
endif()

target_compile_definitions(UpdateTestChild PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
